async function cargarProductos() { const container1 = document.getElementById('listaProductosCard'); const container2 = document.getElementById('listaSeleccionProductos'); if (!_supabase) return; try { const { data, error } = await _supabase.from('productos').select('*'); if (error) { console.error('Error al traer productos de Supabase:', error.message); if (container1) container1.innerHTML = `
Error de conexión: ${error.message}
`; return; } if (!data || data.length === 0) { _productosCache = []; if (container1) container1.innerHTML = '
Sin productos registrados.
'; if (container2) container2.innerHTML = '
No hay productos disponibles.
'; return; } _productosCache = data; if (container1) { container1.innerHTML = _productosCache.map(prod => `
${prod.url_foto ? `` : '
'}
${prod.nombre || 'Sin nombre'}
${prod.notas ? `
📌 ${prod.notas}
` : ''}
${parseFloat(prod.precio || 0).toFixed(2)} €
`).join(''); } if (container2) { container2.innerHTML = _productosCache.map(prod => `
${prod.url_foto ? `` : '
'}
${prod.nombre || 'Sin nombre'}
${prod.notas ? `
📌 ${prod.notas}
` : ''}
${parseFloat(prod.precio || 0).toFixed(2)} €
`).join(''); } } catch(e) { console.error('Error inesperado al cargar productos:', e); } }
