async function cargarProductos() { const container1 = document.getElementById('listaProductosCard'); const container2 = document.getElementById('listaSeleccionProductos'); if (!_supabase) return; try { // Consultamos la tabla de productos const { data, error } = await _supabase.from('productos').select('*'); if (error) { console.error('Error al traer productos:', error.message); return; } if (!data || data.length === 0) { _productosCache = []; container1.innerHTML = '
Sin productos registrados.
'; container2.innerHTML = '
No hay productos disponibles.
'; return; } _productosCache = data; // Renderizar en la pestaña Catálogo container1.innerHTML = _productosCache.map(prod => `
${prod.url_foto ? `` : '
'}
${prod.nombre}
${parseFloat(prod.precio || 0).toFixed(2)} €
`).join(''); // Renderizar en la pestaña Crear Lista container2.innerHTML = _productosCache.map(prod => `
${prod.url_foto ? `` : '
'}
${prod.nombre}
${parseFloat(prod.precio || 0).toFixed(2)} €
`).join(''); } catch(e) { console.error('Error inesperado al cargar productos:', e); } }
