async function cargarProductos() { const c1 = document.getElementById('listaProductosCard'); const c2 = document.getElementById('listaSeleccionProductos'); if (!_supabase) return; try { // Timeout de 8 segundos por si la red no responde const timeoutPromise = new Promise((_, reject) => setTimeout(() => reject(new Error('Tiempo de espera agotado al conectar con Supabase')), 8000) ); const queryPromise = _supabase.from('productos').select('*').order('id', { ascending: false }); const { data, error } = await Promise.race([queryPromise, timeoutPromise]); if (error) { c1.innerHTML = `
⚠️ Error de Supabase: ${error.message}
`; c2.innerHTML = '
Error al obtener productos.
'; return; } if (!data || data.length === 0) { _productosCache = []; c1.innerHTML = '
Sin productos registrados todavía.
'; c2.innerHTML = '
No hay productos disponibles.
'; return; } _productosCache = data; c1.innerHTML = _productosCache.map(p => `
${p.url_foto ? `` : '
📦
'}
${p.nombre || ''}
${p.notas ? `
📌 ${p.notas}
` : ''}
${parseFloat(p.precio || 0).toFixed(2)} €
`).join(''); c2.innerHTML = _productosCache.map(p => `
${p.url_foto ? `` : '
📦
'}
${p.nombre || ''}
${p.notas ? `
📌 ${p.notas}
` : ''}
${parseFloat(p.precio || 0).toFixed(2)} €
`).join(''); } catch(e) { c1.innerHTML = `
⚠️ Error: ${e.message}
`; c2.innerHTML = '
Error al conectar.
'; } }
